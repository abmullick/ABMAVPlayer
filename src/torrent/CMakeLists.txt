cmake_minimum_required(VERSION 3.21)
project(abm_torrent_engine LANGUAGES CXX)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
add_library(abm_torrent_engine SHARED torrent_engine.cpp)
target_compile_features(abm_torrent_engine PRIVATE cxx_std_17)
target_compile_definitions(abm_torrent_engine PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(abm_torrent_engine PRIVATE /EHsc /utf-8 /W4)
endif()
target_include_directories(abm_torrent_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(abm_torrent_engine PRIVATE LibtorrentRasterbar::torrent-rasterbar ws2_32 bcrypt)
set_target_properties(abm_torrent_engine PROPERTIES OUTPUT_NAME "abm_torrent_engine" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
